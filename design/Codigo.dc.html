1//! ETL de ecommerce: catalogo + precios -> Postgres.
2//! Un binario, un DAG. Se despliega copiandolo a pipelines/.
3
4use zeruel::{dag, pipeline, Result};
5
6mod tareas;
7use tareas::*;
8
9fn main() -> Result<()> {
10 let dag = dag!("ecommerce-etl",
11 "bajar-catalogo" => [],
12 "bajar-precios" => [],
13 "limpiar" => ["bajar-catalogo", "bajar-precios"],
14 "enriquecer" => ["limpiar"],
15 "cargar" => ["enriquecer"],
16 );
17
18 pipeline(dag)
19 .cron("0 4 * * *")
20 .techo_ram("2G")
21 // El id de cada tarea sale del propio Task: si no coincide
22 // con el dag!, falla al construir y no al ejecutar.
23 .register(BajarCatalogo::new())
24 .register(BajarPrecios::new())
25 .register(Limpiar::new())
26 .register(Enriquecer::new().reintentos(3))
27 .register(Cargar::new())
28 .start()
29}
